#include <stdio.h>

int main()
{
    int p[2][2], c[2][2];
    int det, inv;
    int k11, k12, k21, k22;
    int i;

    printf("Enter plaintext matrix (2 x 2):\n");
    scanf("%d %d", &p[0][0], &p[0][1]);
    scanf("%d %d", &p[1][0], &p[1][1]);

    printf("Enter ciphertext matrix (2 x 2):\n");
    scanf("%d %d", &c[0][0], &c[0][1]);
    scanf("%d %d", &c[1][0], &c[1][1]);

    det = (p[0][0] * p[1][1] -
           p[0][1] * p[1][0]) % 26;

    if (det < 0)
        det = det + 26;

    inv = 0;

    for (i = 1; i < 26; i++)
    {
        if ((det * i) % 26 == 1)
        {
            inv = i;
            break;
        }
    }

    if (inv == 0)
    {
        printf("Key cannot be recovered.\n");
        printf("Choose a plaintext matrix with determinant relatively prime to 26.\n");
        return 0;
    }

    /* K = C x inverse(P) */

    k11 = (c[0][0] * p[1][1] -
           c[0][1] * p[1][0]) * inv;

    k12 = (-c[0][0] * p[0][1] +
            c[0][1] * p[0][0]) * inv;

    k21 = (c[1][0] * p[1][1] -
           c[1][1] * p[1][0]) * inv;

    k22 = (-c[1][0] * p[0][1] +
            c[1][1] * p[0][0]) * inv;

    k11 = k11 % 26;
    k12 = k12 % 26;
    k21 = k21 % 26;
    k22 = k22 % 26;

    if (k11 < 0) k11 += 26;
    if (k12 < 0) k12 += 26;
    if (k21 < 0) k21 += 26;
    if (k22 < 0) k22 += 26;

    printf("\nRecovered Key Matrix:\n");
    printf("%d %d\n", k11, k12);
    printf("%d %d\n", k21, k22);

    return 0;
}
