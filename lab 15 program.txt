#include <stdio.h>

int main()
{
    char c[200];
    int shift, i, rank;

    printf("Enter ciphertext: ");
    fgets(c, 200, stdin);

    printf("\nPossible plaintexts:\n");

    for (rank = 1; rank <= 10; rank++)
    {
        shift = rank - 1;

        printf("%d. ", rank);

        for (i = 0; c[i] != '\0'; i++)
        {
            if (c[i] >= 'A' && c[i] <= 'Z')
                printf("%c", (c[i] - 'A' - shift + 26) % 26 + 'A');
            else if (c[i] >= 'a' && c[i] <= 'z')
                printf("%c", (c[i] - 'a' - shift + 26) % 26 + 'a');
            else
                printf("%c", c[i]);
        }
    }

    return 0;
}
