
#include <stdio.h>
#include <string.h>

int gcd(int a, int b)
{
    while (b != 0)
    {
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

int main()
{
    char text[100];
    int a, b, i;

    printf("Enter plaintext (uppercase): ");
    scanf("%99s", text);

    printf("Enter a and b: ");
    scanf("%d%d", &a, &b);

    if (a < 0 || gcd(a, 26) != 1)
    {
        printf("Invalid a value\n");
        return 0;
    }

    printf("Ciphertext: ");
    for (i = 0; text[i] != '\0'; i++)
    {
        if (text[i] >= 'A' && text[i] <= 'Z')
            printf("%c", ((a * (text[i] - 'A') + b) % 26) + 'A');
        else
            printf("%c", text[i]);
    }

    printf("\n");
    return 0;
}
