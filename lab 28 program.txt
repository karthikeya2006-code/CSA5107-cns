#include <stdio.h>

int main()
{
    int q = 23, a = 5;
    int x = 6, y = 15;
    int A = 1, B = 1, i;
    int key1 = 1, key2 = 1;

    for (i = 0; i < x; i++)
        A = (A * a) % q;

    for (i = 0; i < y; i++)
        B = (B * a) % q;

    for (i = 0; i < x; i++)
        key1 = (key1 * B) % q;

    for (i = 0; i < y; i++)
        key2 = (key2 * A) % q;

    printf("Alice sends: %d\n", A);
    printf("Bob sends: %d\n", B);
    printf("Alice's shared key: %d\n", key1);
    printf("Bob's shared key: %d\n", key2);

    return 0;
}
