
#include <stdio.h>

int mod26(int x)
{
    return (x % 26 + 26) % 26;
}

int inverse(int x)
{
    int i;
    for (i = 1; i < 26; i++)
    {
        if (mod26(x * i) == 1)
            return i;
    }
    return -1;
}

int main()
{
    int p[2][2], c[2][2], inv[2][2], k[2][2];
    int det, d, i, j, r, t;

    printf("Enter plaintext matrix (2x2):\n");
    for (i = 0; i < 2; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &p[i][j]);

    printf("Enter ciphertext matrix (2x2):\n");
    for (i = 0; i < 2; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &c[i][j]);

    det = mod26(p[0][0] * p[1][1] -
                p[0][1] * p[1][0]);

    d = inverse(det);

    if (d == -1)
    {
        printf("Plaintext matrix is not invertible.\n");
        return 0;
    }

    inv[0][0] = mod26(d * p[1][1]);
    inv[0][1] = mod26(-d * p[0][1]);
    inv[1][0] = mod26(-d * p[1][0]);
    inv[1][1] = mod26(d * p[0][0]);

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
        {
            t = 0;
            for (r = 0; r < 2; r++)
                t += c[i][r] * inv[r][j];

            k[i][j] = mod26(t);
        }
    }

    printf("Recovered key matrix:\n");
    for (i = 0; i < 2; i++)
        printf("%d %d\n", k[i][0], k[i][1]);

    return 0;
}
