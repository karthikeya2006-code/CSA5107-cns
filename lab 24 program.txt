#include <stdio.h>

int main()
{
    int n = 3599, e = 31;
    int p = 59, q = 61;
    int phi = (p - 1) * (q - 1);
    int d;

    for (d = 1; d < phi; d++)
    {
        if ((e * d) % phi == 1)
            break;
    }

    printf("p = %d\n", p);
    printf("q = %d\n", q);
    printf("Phi(n) = %d\n", phi);
    printf("Public key: (%d, %d)\n", e, n);
    printf("Private key: (%d, %d)\n", d, n);

    return 0;
}
