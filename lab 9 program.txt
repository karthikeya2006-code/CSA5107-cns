#include <stdio.h>
#include <ctype.h>

char matrix[5][5];

int main()
{
    char key[] = "ROYAL NEW ZEALAND NAVY";
    char cipher[] =
    "KXJEY UREBE ZWEHE WRYTU HEYFS "
    "KREHE GOYFI WTTTU OLKSY CAJPO "
    "BOTEI ZONTX BYBNT GONEY CUZWR "
    "GDSON SXBOU YWRHE BAAHY USEDQ";

    char alphabet[] = "ABCDEFGHIKLMNOPQRSTUVWXYZ";
    char text[200];
    int used[26] = {0};
    int i, j, k, n;
    int r1, c1, r2, c2;
    char ch;

    k = 0;

    for (i = 0; key[i] != '\0'; i++)
    {
        ch = toupper(key[i]);

        if (ch == 'J')
            ch = 'I';

        if (ch >= 'A' && ch <= 'Z' && !used[ch - 'A'])
        {
            matrix[k / 5][k % 5] = ch;
            used[ch - 'A'] = 1;
            k++;
        }
    }

    for (i = 0; alphabet[i] != '\0'; i++)
    {
        ch = alphabet[i];

        if (!used[ch - 'A'])
        {
            matrix[k / 5][k % 5] = ch;
            used[ch - 'A'] = 1;
            k++;
        }
    }

    n = 0;

    for (i = 0; cipher[i] != '\0'; i++)
    {
        if (cipher[i] >= 'A' && cipher[i] <= 'Z')
        {
            text[n] = cipher[i];
            n++;
        }
    }

    printf("Decrypted message: ");

    for (i = 0; i < n; i += 2)
    {
        r1 = c1 = r2 = c2 = 0;

        for (j = 0; j < 25; j++)
        {
            if (matrix[j / 5][j % 5] == text[i])
            {
                r1 = j / 5;
                c1 = j % 5;
            }

            if (matrix[j / 5][j % 5] == text[i + 1])
            {
                r2 = j / 5;
                c2 = j % 5;
            }
        }

        if (r1 == r2)
        {
            printf("%c", matrix[r1][(c1 + 4) % 5]);
            printf("%c", matrix[r2][(c2 + 4) % 5]);
        }
        else if (c1 == c2)
        {
            printf("%c", matrix[(r1 + 4) % 5][c1]);
            printf("%c", matrix[(r2 + 4) % 5][c2]);
        }
        else
        {
            printf("%c", matrix[r1][c2]);
            printf("%c", matrix[r2][c1]);
        }
    }

    printf("\n");

    return 0;
}
