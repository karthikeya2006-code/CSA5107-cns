#include <stdio.h>
#include <ctype.h>

int main()
{
    char key[] = "CIPHER";
    char cipher[26];
    char text[100];
    int used[26] = {0};
    int i, j, k = 0;

    /* Add keyword letters */
    for (i = 0; key[i] != '\0'; i++)
    {
        char ch = toupper(key[i]);

        if (used[ch - 'A'] == 0)
        {
            cipher[k] = ch;
            used[ch - 'A'] = 1;
            k++;
        }
    }

    /* Add remaining letters */
    for (i = 0; i < 26; i++)
    {
        if (used[i] == 0)
        {
            cipher[k] = 'A' + i;
            k++;
        }
    }

    printf("Cipher alphabet: ");
    for (i = 0; i < 26; i++)
        printf("%c", cipher[i]);

    printf("\nEnter plaintext: ");
    scanf(" %[^\n]", text);

    printf("Ciphertext: ");

    for (i = 0; text[i] != '\0'; i++)
    {
        if (text[i] >= 'a' && text[i] <= 'z')
            printf("%c", cipher[text[i] - 'a']);
        else if (text[i] >= 'A' && text[i] <= 'Z')
            printf("%c", cipher[text[i] - 'A']);
        else
            printf("%c", text[i]);
    }

    return 0;
}
