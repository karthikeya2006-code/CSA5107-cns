
#include <stdio.h>
#include <string.h>

int main()
{
    char plaintext[100];
    int key[100], i, n;

    printf("Enter plaintext (uppercase): ");
    scanf("%99s", plaintext);

    n = strlen(plaintext);

    printf("Enter %d key values (0 to 26): ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &key[i]);

    printf("Ciphertext: ");
    for (i = 0; i < n; i++)
    {
        if (plaintext[i] < 'A' || plaintext[i] > 'Z'
            || key[i] < 0 || key[i] > 26)
        {
            printf("\nInvalid input");
            return 0;
        }

        printf("%c", ((plaintext[i] - 'A' + key[i]) % 26) + 'A');
    }

    printf("\n");
    return 0;
}
