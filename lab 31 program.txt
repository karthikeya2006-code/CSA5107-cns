#include <stdio.h>
#include <stdint.h>

uint64_t shift64(uint64_t x) {
    uint64_t r = x << 1;
    if (x >> 63)
        r ^= 0x1B;
    return r;
}

int main() {
    uint64_t L, K1, K2;

    printf("Enter 64-bit hexadecimal L: ");
    scanf("%llx", (unsigned long long *)&L);

    K1 = shift64(L);
    K2 = shift64(K1);

    printf("K1 = %016llX\n", (unsigned long long)K1);
    printf("K2 = %016llX\n", (unsigned long long)K2);

    return 0;
}
