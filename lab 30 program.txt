#include <stdio.h>

int main()
{
    int X, T, Y, MAC;

    printf("Enter message block X: ");
    scanf("%d", &X);

    printf("Enter MAC value T: ");
    scanf("%d", &T);

    Y = X ^ T;
    MAC = Y ^ T;

    printf("Second block (X XOR T) = %d\n", Y);
    printf("CBC-MAC of two blocks = %d\n", MAC);

    return 0;
}
