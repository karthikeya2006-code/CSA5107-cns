#include <stdio.h>
#include <ctype.h>

char matrix[5][5] = {
    {'M','F','H','I','K'},
    {'U','N','O','P','Q'},
    {'Z','V','W','X','Y'},
    {'E','L','A','R','G'},
    {'D','S','T','B','C'}
};

void findPosition(char ch, int *row, int *col)
{
    int i, j;

    if (ch == 'J')
        ch = 'I';

    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 5; j++)
        {
            if (matrix[i][j] == ch)
            {
                *row = i;
                *col = j;
                return;
            }
        }
    }
}

int main()
{
    char message[] = "MUST SEE YOU OVER CADOGAN WEST COMING AT ONCE";
    char text[200];
    int i, n = 0;
    int r1, c1, r2, c2;

    /* Remove spaces and convert to uppercase */
    for (i = 0; message[i] != '\0'; i++)
    {
        if (isalpha(message[i]))
        {
            text[n] = toupper(message[i]);

            if (text[n] == 'J')
                text[n] = 'I';

            n++;
        }
    }

    /* Add X between repeated letters */
    for (i = 0; i < n - 1; i = i + 2)
    {
        if (text[i] == text[i + 1])
        {
            int j;

            for (j = n; j > i + 1; j--)
                text[j] = text[j - 1];

            text[i + 1] = 'X';
            n++;
        }
    }

    /* Add X if length is odd */
    if (n % 2 != 0)
    {
        text[n] = 'X';
        n++;
    }

    printf("Encrypted message: ");

    /* Encrypt pairs */
    for (i = 0; i < n; i = i + 2)
    {
        findPosition(text[i], &r1, &c1);
        findPosition(text[i + 1], &r2, &c2);

        /* Same row */
        if (r1 == r2)
        {
            printf("%c", matrix[r1][(c1 + 1) % 5]);
            printf("%c", matrix[r2][(c2 + 1) % 5]);
        }

        /* Same column */
        else if (c1 == c2)
        {
            printf("%c", matrix[(r1 + 1) % 5][c1]);
            printf("%c", matrix[(r2 + 1) % 5][c2]);
        }

        /* Rectangle rule */
        else
        {
            printf("%c", matrix[r1][c2]);
            printf("%c", matrix[r2][c1]);
        }

        printf(" ");
    }

    return 0;
}
