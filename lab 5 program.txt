#include <stdio.h>
#include <ctype.h>

int main()
{
    char text[100];
    int a, b, i, p, c;

    printf("Enter plaintext: ");
    scanf(" %[^\n]", text);

    printf("Enter a: ");
    scanf("%d", &a);

    printf("Enter b: ");
    scanf("%d", &b);

    printf("Ciphertext: ");

    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
        {
            p = toupper(text[i]) - 'A';
            c = (a * p + b) % 26;
            printf("%c", c + 'A');
        }
        else
        {
            printf("%c", text[i]);
        }
    }

    return 0;
}
