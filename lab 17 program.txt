
#include <stdio.h>

int main()
{
    int K[16], i, temp;

    printf("Enter 16 encryption keys (K1 to K16):\n");

    for (i = 0; i < 16; i++)
        scanf("%d", &K[i]);

    printf("\nDES Decryption Key Order:\n");

    for (i = 15; i >= 0; i--)
        printf("K%d = %d\n", i + 1, K[i]);

    return 0;
}
